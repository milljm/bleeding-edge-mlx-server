# Apple Silicon MLX stack for the `edge` conda env.
#
#   conda create -n edge python=3.13 uv pip git
#   conda activate edge
#   git clone https://github.com/milljm/bleeding-edge-mlx-server.git
#   cd bleeding-edge-mlx-server
#   uv pip install -r requirements.txt
#   mlx-edge serve
#
# `mlx` is the compiled Metal wheel from PyPI (3.13 macOS arm64 wheels exist).
# mlx-lm / mlx-vlm / mlx-audio install from git HEAD so new architectures land
# without waiting on a release. `-e .` is this repo's mlx-edge CLI.
#
# Extras mlx-vlm has not declared (and may not, until those PRs merge):
#   safetensors / torchvision — GLM-5 Next vision tower imports torchvision
#     (that pulls torch). Apple Silicon wheels, not the CUDA SD conda stack.
#   mlx-audio — mlx_vlm.server --tts-model / --stt-model import this. Overlay
#     with `mlx-edge build audio` the same way as mlx-vlm.
